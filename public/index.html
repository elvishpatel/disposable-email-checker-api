      Why EmailGuard?
      
    
      
      
      
        
      
        
      9,241 Blocked Domains
      
        
      Merged from three community blocklists — nearly 2× larger than v1, auto-updated with every redeploy.
      
      
      
      
      
        
      
        
      Confidence Scoring
      
        
      0–100 score with label (very_low → very_high) across 8 independent signals: blocklist, MX, DNS, pattern analysis, and more.
      
      
      
      
      
        
      
        
      DNS + MX Verification
      
        
      Checks if the domain actually exists and can receive email, catching invalid domains that aren't yet in the blocklist.
      
      
      
      
      
        
      
        
      Bulk Endpoint
      
        
      Verify up to 50 emails in a single request with a per-request summary — great for batch cleanups.
      
      
      
      
      
        
      
        
      Proper Rate Limiting
      
        
      In-memory 1-hour sliding window with correct RFC headers: X-RateLimit-Limit, Remaining, Reset.
      
      
      
      
      
        
      
        
      Clean JSON API
      
        
      Consistent response shape across all endpoints. No SDK required — works with any HTTP client.
      
      
      
    
      
  
      

      



      
  
      
    
      API Reference
      

    
    
      
      
      
        POST
        https://email-validator-api-uk66.onrender.com/v1/verify
      
      
      
      Validate a single email address. Returns verdict, confidence score, DNS info, and reasons.
      

      
      
        
      Request Body
      
        
      
          
            
      ParameterTypeRequiredDescription
      
                
          
            
      
              email
              string
              required
              Email address to validate
            
      
            
      
              check_dns
              boolean
              optional
              Enable DNS + MX lookup. Default: true
            
      
                
        
      
      
      

      
      
        
      Response Headers
      
        
      
          
      HeaderDescription
      
          
            
      X-RateLimit-LimitMax requests per hour for your tier
      
            
      X-RateLimit-RemainingRequests left in current window
      
            
      X-RateLimit-ResetUnix epoch when the window resets
      
            
      Retry-AfterSeconds to wait (only on 429)

      import java.net.URI;
import java.net.http.*;

var client = HttpClient.newHttpClient();
var req = HttpRequest.newBuilder()
    .uri(URI.create("https://email-validator-api-uk66.onrender.com/v1/verify"))
    .header("Content-Type", "application/json")
    .POST(HttpRequest.BodyPublishers.ofString("{\"email\":\"test@mailinator.com\"}"))
    .build();
var resp = client.send(req, HttpResponse.BodyHandlers.ofString());
System.out.println(resp.body());

        POST
        /v1/verify/bulk
      
      
      
      Verify up to 50 emails in one request. Returns a summary + per-email results array.
      
      
      
        
      Request Body
      
        
      
          
      ParameterTypeRequiredDescription
      
          
            
      emailsstring[]requiredArray of email strings (max 50)
      
            
      check_dnsbooleanoptionalDNS check per email. Default: false (for speed)
      
                
        
      
      
      
      
      
        
      Response Example
      
        
      
          
          
      {
  "status": "ok",
  "summary": {
    "total": 3,
    "valid": 1,
    "invalid": 1,
    "risky": 1
  },
  "results": [ /* array of full verify objects */ ]
}
      
        
      
      
      
    
      

    
    
      
      
      
        GET
        /v1/domain/:domain
      
      
      
      Check a domain directly without specifying the local part. Always performs DNS lookup.
      
      
      
        
      Response Example
      
        
      
          
          
      {
  "status": "ok",
  "domain": "mailinator.com",
  "is_disposable": true,
  "confidence": { "score": 80, "label": "very_high" },
  "risk_level": "high",
  "reasons": [ "Domain is in the known disposable email blocklist" ],
  "dns": {
    "domain_exists": true,
    "has_mx": true,
    "mx_records": [{ "priority": 10, "exchange": "mail.mailinator.com" }],
    "primary_mx": "mail.mailinator.com"
  }
}

  "remaining": 38,
